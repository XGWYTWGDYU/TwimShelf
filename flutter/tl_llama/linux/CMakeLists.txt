cmake_minimum_required(VERSION 3.21)
set(PROJECT_NAME "tl_llama")
project(${PROJECT_NAME} LANGUAGES C CXX)

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../src" "${CMAKE_CURRENT_BINARY_DIR}/shared")

set(tl_llama_bundled_libraries "" PARENT_SCOPE)
set(_files "")
foreach(t ${TL_LLAMA_LIBRARIES})
  list(APPEND _files $<TARGET_FILE:${t}>)
  get_target_property(_type ${t} TYPE)
  get_target_property(_version ${t} VERSION)
  get_target_property(_soversion ${t} SOVERSION)
  if(_type STREQUAL "SHARED_LIBRARY")
    # Flutter copies the files without relinking. Keep the loader name and sibling lookup.
    set_property(TARGET ${t} APPEND PROPERTY BUILD_RPATH "$ORIGIN")
    set_property(TARGET ${t} PROPERTY BUILD_RPATH_USE_ORIGIN ON)
    if(NOT _soversion MATCHES "-NOTFOUND$" AND NOT _version STREQUAL _soversion)
      list(APPEND _files $<TARGET_SONAME_FILE:${t}>)
    endif()
  endif()
endforeach()
set(tl_llama_bundled_libraries ${_files} PARENT_SCOPE)
