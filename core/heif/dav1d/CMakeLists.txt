# dav1d's own build is Meson, which none of the targets otherwise need. The C code alone is built
# here, the same way everywhere: no assembly, slower than dav1d at its best but it runs on every target.
cmake_minimum_required(VERSION 3.16)
project(dav1d VERSION 1.5.3 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(TO_CMAKE_PATH "${DAV1D_SOURCE}" S)

foreach(v ARCH_AARCH64 ARCH_ARM ARCH_X86 ARCH_X86_32 ARCH_X86_64 ARCH_PPC64LE ARCH_RISCV ARCH_RV32 ARCH_RV64 ARCH_LOONGARCH ARCH_LOONGARCH32 ARCH_LOONGARCH64
        ENDIANNESS_BIG HAVE_ASM TRIM_DSP_FUNCTIONS CONFIG_LOG CONFIG_MACOS_KPERF HAVE_DLSYM HAVE_MEMALIGN HAVE_ALIGNED_ALLOC HAVE_GETAUXVAL HAVE_ELF_AUX_INFO
        HAVE_PTHREAD_NP_H HAVE_PTHREAD_GETAFFINITY_NP HAVE_PTHREAD_SETAFFINITY_NP HAVE_PTHREAD_SETNAME_NP HAVE_PTHREAD_SET_NAME_NP HAVE_IO_H)
  set(config "${config}#define ${v} 0\n")
endforeach()
set(config "${config}#define CONFIG_8BPC 1\n#define CONFIG_16BPC 1\n#define HAVE_SYS_TYPES_H 1\n")
if(WIN32)
  set(config "${config}#define HAVE_CLOCK_GETTIME 0\n#define HAVE_POSIX_MEMALIGN 0\n#define HAVE_UNISTD_H 0\n#define HAVE_C11_GENERIC 0\n#define _WIN32_WINNT 0x0601\n#define UNICODE 1\n#define _UNICODE 1\n#define _CRT_DECLARE_NONSTDC_NAMES 1\n")
else()
  set(config "${config}#define HAVE_CLOCK_GETTIME 1\n#define HAVE_POSIX_MEMALIGN 1\n#define HAVE_UNISTD_H 1\n#define HAVE_C11_GENERIC 1\n#define _GNU_SOURCE 1\n#define _DARWIN_C_SOURCE 1\n")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/gen/config.h" "${config}")
file(WRITE "${CMAKE_BINARY_DIR}/gen/vcs_version.h" "#define DAV1D_VERSION \"1.5.3\"\n")
file(WRITE "${CMAKE_BINARY_DIR}/gen/dav1d/version.h" "#define DAV1D_API_VERSION_MAJOR 7\n#define DAV1D_API_VERSION_MINOR 0\n#define DAV1D_API_VERSION_PATCH 0\n")

set(sources cdf cpu ctx data decode dequant_tables getbits intra_edge itx_1d lf_mask lib log mem msac obu pal picture qm ref refmvs scan tables thread_task warpmv wedge)
list(TRANSFORM sources PREPEND "${S}/src/")
list(TRANSFORM sources APPEND ".c")
if(WIN32)
  list(APPEND sources "${S}/src/win32/thread.c")
endif()
set(templates cdef_apply cdef fg_apply filmgrain ipred_prepare ipred itx lf_apply loopfilter looprestoration lr_apply mc recon)
list(TRANSFORM templates PREPEND "${S}/src/")
list(TRANSFORM templates APPEND "_tmpl.c")

set(includes "${CMAKE_BINARY_DIR}/gen" "${S}" "${S}/include" "${CMAKE_BINARY_DIR}/gen/dav1d")
if(MSVC)
  list(APPEND includes "${S}/include/compat/msvc")
endif()
foreach(depth 8 16)
  add_library(dav1d_${depth} OBJECT ${templates})
  target_compile_definitions(dav1d_${depth} PRIVATE BITDEPTH=${depth})
  target_include_directories(dav1d_${depth} PRIVATE ${includes})
endforeach()
add_library(dav1d STATIC ${sources} $<TARGET_OBJECTS:dav1d_8> $<TARGET_OBJECTS:dav1d_16>)
target_include_directories(dav1d PRIVATE ${includes})

install(TARGETS dav1d ARCHIVE DESTINATION lib)
install(DIRECTORY "${S}/include/dav1d" DESTINATION include FILES_MATCHING PATTERN "*.h")
install(FILES "${CMAKE_BINARY_DIR}/gen/dav1d/version.h" DESTINATION include/dav1d)
